public class CalcDemo {
    // 两数求和方法
    public static int calculateAdd(int a, int b) {
        return a + b;
    }

    public static void main(String[] args) {
        int first = 35;
        int second = 65;
        int sumRes = calculateAdd(first, second);
        System.out.println("两数相加结果：" + sumRes);

        // 简单for循环遍历
        for (int i = 1; i <= 4; i++) {
            System.out.println("当前循环轮次：" + i);
        }
    }
}

public class CalcTest {
    // 整数相加运算
    public static int getSum(int m, int n) {
        int total = m + n;
        return total;
    }

    public static void main(String[] args) {
        int numOne = 35;
        int numTwo = 65;
        int result = getSum(numOne, numTwo);
        System.out.println("数值相加总和：" + result);

        // 循环打印序号
        for (int k = 1; k <= 4; k++) {
            System.out.println("当前遍历序号：" + k);
        }
    }
}
